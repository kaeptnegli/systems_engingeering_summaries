	\subsection{Strahlensatz}
	
	\begin{tcolorbox}[dglbox, title={Strahlensatz}]
		Werden zwei Geraden von zwei Parallelen geschnitten, so verhalten sich
		die entstehenden Abschnitte proportional:
		\[
		\frac{a}{a'} = \frac{b}{b'} = \frac{c}{c'}
		\]
	\end{tcolorbox}
