	\subsubsection{Fundamentale Grenzwerte}
	
	\begin{tcolorbox}[dglbox, title={Wichtige Grenzwerte}]
		\[
		\lim_{x\to 0} \frac{\sin x}{x} = 1
		\qquad\qquad
		\lim_{x\to 0} \frac{\cos x - 1}{x} = 0
		\]
	\end{tcolorbox}
	
	\paragraph{Beweis: $\lim_{x\to 0}\dfrac{\sin x}{x}=1$ \quad (Sandwich-Satz)}
	
	Betrachte den Einheitskreis und einen Winkel $x>0$ im Bogenmass.
	Geometrisch gilt:
	\[
	\sin x \le x \le \tan x = \frac{\sin x}{\cos x}.
	\]
	Division durch $\sin x > 0$ und anschliessende Kehrwertbildung liefert:
	\[
	\cos x \le \frac{\sin x}{x} \le 1.
	\]
	Da $\cos x \to 1$ für $x\to 0$, folgt mit dem Sandwich-Satz:
	\[
	\lim_{x\to 0}\frac{\sin x}{x} = 1.
	\]
	
	\paragraph{Beweis: $\lim_{x\to 0}\dfrac{\cos x-1}{x}=0$}
	
	Mit der Identität $\cos x = 1-2\sin^2\!\left(\tfrac{x}{2}\right)$
	(siehe Abschnitt Trigonometrie) folgt:
	\[
	\frac{\cos x-1}{x}
	= \frac{-2\sin^2(x/2)}{x}
	= -\frac{\sin(x/2)}{x/2}\cdot\sin\!\left(\tfrac{x}{2}\right)
	\;\xrightarrow{x\to 0}\; -1\cdot 0 = 0.
	\]
